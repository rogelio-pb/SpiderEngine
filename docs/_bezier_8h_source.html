20 // Performs a cubic bezier interpolation between four control points,
21 // returning the value at the specified time (t ranges 0 to 1).
22 template<typename T>
23 inline T CubicInterpolate(T const& p1, T const& p2, T const& p3, T const& p4, float t) noexcept
24 {
25 return p1 * (1 - t) * (1 - t) * (1 - t) +
26 p2 * 3 * t * (1 - t) * (1 - t) +
27 p3 * 3 * t * t * (1 - t) +
28 p4 * t * t * t;
29 }
30
31 template<>
32 inline DirectX::XMVECTOR CubicInterpolate(DirectX::XMVECTOR const& p1, DirectX::XMVECTOR const& p2, DirectX::XMVECTOR const& p3, DirectX::XMVECTOR const& p4, float t) noexcept
33 {
34 using namespace DirectX;
35
36 const XMVECTOR T0 = XMVectorReplicate((1 - t) * (1 - t) * (1 - t));
37 const XMVECTOR T1 = XMVectorReplicate(3 * t * (1 - t) * (1 - t));
38 const XMVECTOR T2 = XMVectorReplicate(3 * t * t * (1 - t));
39 const XMVECTOR T3 = XMVectorReplicate(t * t * t);
40
41 XMVECTOR Result = XMVectorMultiply(p1, T0);
42 Result = XMVectorMultiplyAdd(p2, T1, Result);
43 Result = XMVectorMultiplyAdd(p3, T2, Result);
44 Result = XMVectorMultiplyAdd(p4, T3, Result);
45
46 return Result;
47 }

50 // Computes the tangent of a cubic bezier curve at the specified time.
51 template<typename T>
52 inline T CubicTangent(T const& p1, T const& p2, T const& p3, T const& p4, float t) noexcept
53 {
54 return p1 * (-1 + 2 * t - t * t) +
55 p2 * (1 - 4 * t + 3 * t * t) +
56 p3 * (2 * t - 3 * t * t) +
57 p4 * (t * t);
58 }
59
60 template<>
61 inline DirectX::XMVECTOR CubicTangent(DirectX::XMVECTOR const& p1, DirectX::XMVECTOR const& p2, DirectX::XMVECTOR const& p3, DirectX::XMVECTOR const& p4, float t) noexcept
62 {
63 using namespace DirectX;
64
65 const XMVECTOR T0 = XMVectorReplicate(-1 + 2 * t - t * t);
66 const XMVECTOR T1 = XMVectorReplicate(1 - 4 * t + 3 * t * t);
67 const XMVECTOR T2 = XMVectorReplicate(2 * t - 3 * t * t);
68 const XMVECTOR T3 = XMVectorReplicate(t * t);
69
70 XMVECTOR Result = XMVectorMultiply(p1, T0);
71 Result = XMVectorMultiplyAdd(p2, T1, Result);
72 Result = XMVectorMultiplyAdd(p3, T2, Result);
73 Result = XMVectorMultiplyAdd(p4, T3, Result);
74
75 return Result;
76 }

79 // Creates vertices for a patch that is tessellated at the specified level.
80 // Calls the specified outputVertex function for each generated vertex,
81 // passing the position, normal, and texture coordinate as parameters.
82 template<typename TOutputFunc>
83 void CreatePatchVertices(_In_reads_(16) const DirectX::XMVECTOR patch[16], size_t tessellation, bool isMirrored, TOutputFunc outputVertex)
84 {
85 using namespace DirectX;
86
87 for (size_t i = 0; i <= tessellation; i++)
88 {
89 const float u = float(i) / float(tessellation);
90
91 for (size_t j = 0; j <= tessellation; j++)
92 {
93 const float v = float(j) / float(tessellation);
94
95 // Perform four horizontal bezier interpolations
96 // between the control points of this patch.
97 const XMVECTOR p1 = CubicInterpolate(patch[0], patch[1], patch[2], patch[3], u);
98 const XMVECTOR p2 = CubicInterpolate(patch[4], patch[5], patch[6], patch[7], u);
99 const XMVECTOR p3 = CubicInterpolate(patch[8], patch[9], patch[10], patch[11], u);
100 const XMVECTOR p4 = CubicInterpolate(patch[12], patch[13], patch[14], patch[15], u);
101
102 // Perform a vertical interpolation between the results of the
103 // previous horizontal interpolations, to compute the position.
104 const XMVECTOR position = CubicInterpolate(p1, p2, p3, p4, v);
105
106 // Perform another four bezier interpolations between the control
107 // points, but this time vertically rather than horizontally.
108 const XMVECTOR q1 = CubicInterpolate(patch[0], patch[4], patch[8], patch[12], v);
109 const XMVECTOR q2 = CubicInterpolate(patch[1], patch[5], patch[9], patch[13], v);
110 const XMVECTOR q3 = CubicInterpolate(patch[2], patch[6], patch[10], patch[14], v);
111 const XMVECTOR q4 = CubicInterpolate(patch[3], patch[7], patch[11], patch[15], v);
112
113 // Compute vertical and horizontal tangent vectors.
114 const XMVECTOR tangent1 = CubicTangent(p1, p2, p3, p4, v);
115 const XMVECTOR tangent2 = CubicTangent(q1, q2, q3, q4, u);
116
117 // Cross the two tangent vectors to compute the normal.
118 XMVECTOR normal = XMVector3Cross(tangent1, tangent2);
119
120 if (!XMVector3NearEqual(normal, XMVectorZero(), g_XMEpsilon))
121 {
122 normal = XMVector3Normalize(normal);
123
124 // If this patch is mirrored, we must invert the normal.
125 if (isMirrored)
126 {
127 normal = XMVectorNegate(normal);
128 }
129 }
130 else
131 {
132 // In a tidy and well constructed bezier patch, the preceding
133 // normal computation will always work. But the classic teapot
134 // model is not tidy or well constructed! At the top and bottom
135 // of the teapot, it contains degenerate geometry where a patch
136 // has several control points in the same place, which causes
137 // the tangent computation to fail and produce a zero normal.
138 // We 'fix' these cases by just hard-coding a normal that points
139 // either straight up or straight down, depending on whether we
140 // are on the top or bottom of the teapot. This is not a robust
141 // solution for all possible degenerate bezier patches, but hey,
142 // it's good enough to make the teapot work correctly!
143
144 normal = XMVectorSelect(g_XMIdentityR1, g_XMNegIdentityR1, XMVectorLess(position, XMVectorZero()));
145 }
146
147 // Compute the texture coordinate.
148 const float mirroredU = isMirrored ? 1 - u : u;
149
150 const XMVECTOR textureCoordinate = XMVectorSet(mirroredU, v, 0, 0);
151
152 // Output this vertex.
153 outputVertex(position, normal, textureCoordinate);
154 }
155 }
156 }

159 // Creates indices for a patch that is tessellated at the specified level.
160 // Calls the specified outputIndex function for each generated index value.
161 template<typename TOutputFunc>
162 void CreatePatchIndices(size_t tessellation, bool isMirrored, TOutputFunc outputIndex)
163 {
164 size_t stride = tessellation + 1;
165
166 for (size_t i = 0; i < tessellation; i++)
167 {
168 for (size_t j = 0; j < tessellation; j++)
169 {
170 // Make a list of six index values (two triangles).
171 std::array<size_t, 6> indices =
172 {
173 i * stride + j,
174 (i + 1) * stride + j,
175 (i + 1) * stride + j + 1,
176
177 i * stride + j,
178 (i + 1) * stride + j + 1,
179 i * stride + j + 1,
180 };
181
182 // If this patch is mirrored, reverse indices to fix the winding order.
183 if (isMirrored)
184 {
185 std::reverse(indices.begin(), indices.end());
186 }
187
188 // Output these index values.
189 std::for_each(indices.begin(), indices.end(), outputIndex);
190 }
191 }
192 }
