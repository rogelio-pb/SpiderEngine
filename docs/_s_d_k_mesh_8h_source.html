94
95 D3DDECLTYPE_DEC3N = 14,
96 // 3D signed normalized (v[0]/511.0, v[1]/511.0, v[2]/511.0, 1.)
97 // Note: There is no equivalent to D3DDECLTYPE_DEC3N (14) as a DXGI_FORMAT
98
99 D3DDECLTYPE_FLOAT16_2 = 15,

101
102 D3DDECLTYPE_FLOAT16_4 = 16,
103 // Four 16-bit floating point values
104
105 D3DDECLTYPE_UNUSED = 17,
106 // When the type field in a decl is unused.
107
108 // These are extensions for DXGI-based versions of Direct3D
