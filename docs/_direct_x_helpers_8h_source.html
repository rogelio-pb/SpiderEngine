206 template<typename T>
207 constexpr bool IsPowerOf2(T x) noexcept { return ((x != 0) && !(x & (x - 1))); }

209 // Helpers for aligning values by a power of 2
210 template<typename T>
211 inline T AlignDown(T size, size_t alignment) noexcept
212 {
213 if (alignment > 0)
214 {
215 assert(((alignment - 1) & alignment) == 0);
216 auto mask = static_cast<T>(alignment - 1);
217 return size & ~mask;
218 }
219 return size;
220 }
221
222 template<typename T>
223 inline T AlignUp(T size, size_t alignment) noexcept
224 {
225 if (alignment > 0)
226 {
227 assert(((alignment - 1) & alignment) == 0);
228 auto mask = static_cast<T>(alignment - 1);
229 return (size + mask) & ~mask;
230 }
231 return size;
232 }
233 }
