104 // Similar to the D3D9 API DrawPrimitiveUP.
105 void Draw(
106 D3D11_PRIMITIVE_TOPOLOGY topology,
107 _In_reads_(vertexCount) TVertex const* vertices, size_t vertexCount)
108 {
109 void* mappedVertices;
110
111 PrimitiveBatchBase::Draw(topology, false, nullptr, 0, vertexCount, &mappedVertices);
112
113 memcpy(mappedVertices, vertices, vertexCount * sizeof(TVertex));
114 }
115
116
117 // Similar to the D3D9 API DrawIndexedPrimitiveUP.
118 void DrawIndexed(
119 D3D11_PRIMITIVE_TOPOLOGY topology,
120 _In_reads_(indexCount) uint16_t const* indices, size_t indexCount,
121 _In_reads_(vertexCount) TVertex const* vertices, size_t vertexCount)
122 {
123 void* mappedVertices;
124
125 PrimitiveBatchBase::Draw(topology, true, indices, indexCount, vertexCount, &mappedVertices);
126
