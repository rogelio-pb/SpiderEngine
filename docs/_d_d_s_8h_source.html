57 (static_cast<uint32_t>(static_cast<uint8_t>(ch0)) \
58 | (static_cast<uint32_t>(static_cast<uint8_t>(ch1)) << 8) \
59 | (static_cast<uint32_t>(static_cast<uint8_t>(ch2)) << 16) \
