220 // Helper looks up the bytecode for the specified vertex shader permutation.
221 // Client code needs this in order to create matching input layouts.
222 void GetVertexShaderBytecode(int permutation, _Out_ void const** pShaderByteCode, _Out_ size_t* pByteCodeLength) noexcept
223 {
224 assert(pShaderByteCode != nullptr && pByteCodeLength != nullptr);
225 assert(permutation >= 0 && permutation < Traits::ShaderPermutationCount);
226 _Analysis_assume_(permutation >= 0 && permutation < Traits::ShaderPermutationCount);
227 const int shaderIndex = VertexShaderIndices[permutation];
228 assert(shaderIndex >= 0 && shaderIndex < Traits::VertexShaderCount);
229 _Analysis_assume_(shaderIndex >= 0 && shaderIndex < Traits::VertexShaderCount);
230
231 ShaderBytecode const& bytecode = VertexShaderBytecode[shaderIndex];
232
233 *pShaderByteCode = bytecode.code;
234 *pByteCodeLength = bytecode.length;
235 }

238 // Helper sets our shaders and constant buffers onto the D3D device.
239 void ApplyShaders(_In_ ID3D11DeviceContext* deviceContext, int permutation)
240 {
241 // Set shaders.
242 auto vertexShader = mDeviceResources->GetVertexShader(permutation);
243 auto pixelShader = mDeviceResources->GetPixelShader(permutation);
244
245 deviceContext->VSSetShader(vertexShader, nullptr, 0);
246 deviceContext->PSSetShader(pixelShader, nullptr, 0);
247
248 #if defined(_XBOX_ONE) && defined(_TITLE)
249 void *grfxMemory;
250 mConstantBuffer.SetData(deviceContext, constants, &grfxMemory);
251
252 Microsoft::WRL::ComPtr<ID3D11DeviceContextX> deviceContextX;
253 ThrowIfFailed(deviceContext->QueryInterface(IID_GRAPHICS_PPV_ARGS(deviceContextX.GetAddressOf())));
254
255 auto buffer = mConstantBuffer.GetBuffer();
256
257 deviceContextX->VSSetPlacementConstantBuffer(0, buffer, grfxMemory);
258 deviceContextX->PSSetPlacementConstantBuffer(0, buffer, grfxMemory);
259 #else
260 // Make sure the constant buffer is up to date.
261 if (dirtyFlags & EffectDirtyFlags::ConstantBuffer)
262 {
263 mConstantBuffer.SetData(deviceContext, constants);
264
265 dirtyFlags &= ~EffectDirtyFlags::ConstantBuffer;
266 }
267
268 // Set the constant buffer.
269 ID3D11Buffer* buffer = mConstantBuffer.GetBuffer();
270
271 deviceContext->VSSetConstantBuffers(0, 1, &buffer);
272 deviceContext->PSSetConstantBuffers(0, 1, &buffer);
273 #endif
274 }
