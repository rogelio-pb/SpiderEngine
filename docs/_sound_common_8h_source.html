51 // Helper for getting a format tag from a WAVEFORMATEX
52 inline uint32_t GetFormatTag(const WAVEFORMATEX* wfx) noexcept
53 {
54 if (wfx->wFormatTag == WAVE_FORMAT_EXTENSIBLE)
55 {
56 if (wfx->cbSize < (sizeof(WAVEFORMATEXTENSIBLE) - sizeof(WAVEFORMATEX)))
57 return 0;
58
59 static const GUID s_wfexBase = { 0x00000000, 0x0000, 0x0010, { 0x80, 0x00, 0x00, 0xAA, 0x00, 0x38, 0x9B, 0x71 } };
60
61 auto wfex = reinterpret_cast<const WAVEFORMATEXTENSIBLE*>(wfx);
62
63 if (memcmp(reinterpret_cast<const BYTE*>(&wfex->SubFormat) + sizeof(DWORD),
64 reinterpret_cast<const BYTE*>(&s_wfexBase) + sizeof(DWORD), sizeof(GUID) - sizeof(DWORD)) != 0)
65 {
66 return 0;
67 }
68
69 return wfex->SubFormat.Data1;
70 }
71 else
72 {
73 return wfx->wFormatTag;
74 }
75 }

171 bool Play() // Returns true if STOPPED -> PLAYING
172 {
173 if (voice)
174 {
175 if (state == PAUSED)
176 {
177 HRESULT hr = voice->Start(0);
178 ThrowIfFailed(hr);
179 state = PLAYING;
180 }
181 else if (state != PLAYING)
182 {
183 if (mVolume != 1.f)
184 {
185 HRESULT hr = voice->SetVolume(mVolume);
186 ThrowIfFailed(hr);
187 }
188
189 if (mPitch != 0.f)
190 {
191 mFreqRatio = XAudio2SemitonesToFrequencyRatio(mPitch * 12.f);
192
193 HRESULT hr = voice->SetFrequencyRatio(mFreqRatio);
194 ThrowIfFailed(hr);
195 }
196
197 if (mPan != 0.f)
198 {
199 SetPan(mPan);
200 }
201
202 HRESULT hr = voice->Start(0);
203 ThrowIfFailed(hr);
204 state = PLAYING;
205 return true;
206 }
207 }
208 return false;
209 }

211 void Stop(bool immediate, bool& looped) noexcept
212 {
213 if (!voice)
214 {
215 state = STOPPED;
216 return;
217 }
218
219 if (immediate)
220 {
221 state = STOPPED;
222 std::ignore = voice->Stop(0);
223 std::ignore = voice->FlushSourceBuffers();
224 }
225 else if (looped)
226 {
227 looped = false;
228 std::ignore = voice->ExitLoop();
229 }
230 else
231 {
232 std::ignore = voice->Stop(XAUDIO2_PLAY_TAILS);
233 }
234 }
235
236 void Pause() noexcept
237 {
238 if (voice && state == PLAYING)
239 {
240 state = PAUSED;
241
242 std::ignore = voice->Stop(0);
243 }
244 }
245
246 void Resume()
247 {
248 if (voice && state == PAUSED)
249 {
250 HRESULT hr = voice->Start(0);
251 ThrowIfFailed(hr);
252 state = PLAYING;
253 }
254 }

269 void SetPitch(float pitch)
270 {
271 assert(pitch >= -1.f && pitch <= 1.f);
272
273 if ((mFlags & SoundEffectInstance_NoSetPitch) && pitch != 0.f)
274 {
275 DebugTrace("ERROR: Sound effect instance was created with the NoSetPitch flag\n");
276 throw std::runtime_error("SetPitch");
277 }
278
279 mPitch = pitch;
280
281 if (voice)
282 {
283 mFreqRatio = XAudio2SemitonesToFrequencyRatio(mPitch * 12.f);
284
285 HRESULT hr = voice->SetFrequencyRatio(mFreqRatio);
286 ThrowIfFailed(hr);
287 }

294 SoundState GetState(bool autostop) noexcept
295 {
296 if (autostop && voice && (state == PLAYING))
297 {
298 XAUDIO2_VOICE_STATE xstate;
299 voice->GetState(&xstate, XAUDIO2_VOICE_NOSAMPLESPLAYED);
300
301 if (!xstate.BuffersQueued)
302 {
303 // Automatic stop if the buffer has finished playing
304 std::ignore = voice->Stop();
305 state = STOPPED;
306 }
307 }
308
309 return state;
310 }

312 int GetPendingBufferCount() const noexcept
313 {
314 if (!voice)
315 return 0;
316
317 XAUDIO2_VOICE_STATE xstate;
318 voice->GetState(&xstate, XAUDIO2_VOICE_NOSAMPLESPLAYED);
319 return static_cast<int>(xstate.BuffersQueued);
320 }

351 {

356 voice->DestroyVoice();
357 voice = nullptr;
358 }
359 state = STOPPED;
360 engine = nullptr;
361 mDirectVoice = nullptr;
362 mReverbVoice = nullptr;
363 }

374 void GatherStatistics(AudioStatistics& stats) const noexcept
375 {
376 ++stats.allocatedInstances;

378 {
379 ++stats.allocatedVoices;
380

404 void UpdateCalculateFlags()
405 {
406 assert(engine != nullptr);
407 mX3DCalcFlags = engine->Get3DCalculateFlags();
408 if ((engine->GetChannelMask() & SPEAKER_LOW_FREQUENCY) && (mFlags & SoundEffectInstance_UseRedirectLFE))
409 {
410 mX3DCalcFlags |= X3DAUDIO_CALCULATE_REDIRECT_TO_LFE;
411 }
412
413 if (mFlags & SoundEffectInstance_ZeroCenter3D)
414 {

416 }
