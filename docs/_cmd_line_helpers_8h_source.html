61 T LookupByName(const wchar_t _In_z_ *pName, const SValue<T> *pArray) noexcept
62 {
63 while (pArray->name)
64 {
65 if (!_wcsicmp(pName, pArray->name))
66 return pArray->value;
67
68 pArray++;
69 }
70
71 return static_cast<T>(0);

146 void PrintLogo(bool versionOnly, _In_z_ const wchar_t* name, _In_z_ const wchar_t* desc) noexcept
147 {
148 wchar_t version[32] = {};
149
150 wchar_t appName[_MAX_PATH] = {};
151 if (GetModuleFileNameW(nullptr, appName, _MAX_PATH))
152 {
153 const DWORD size = GetFileVersionInfoSizeW(appName, nullptr);
154 if (size > 0)
155 {
156 auto verInfo = std::make_unique<uint8_t[]>(size);
157 if (GetFileVersionInfoW(appName, 0, size, verInfo.get()))
158 {
159 LPVOID lpstr = nullptr;
160 UINT strLen = 0;
161 if (VerQueryValueW(verInfo.get(), L"\\StringFileInfo\\040904B0\\ProductVersion", &lpstr, &strLen))
162 {
163 wcsncpy_s(version, reinterpret_cast<const wchar_t*>(lpstr), strLen);
164 }
165 }
166 }
167 }
168
169 if (!*version || wcscmp(version, L"1.0.0.0") == 0)
170 {
171 swprintf_s(version, L"%03d (library)", TOOL_VERSION);
172 }
173
174 if (versionOnly)
175 {
176 wprintf(L"%ls version %ls\n", name, version);
177 }
178 else
179 {
180 wprintf(L"%ls Version %ls\n", desc, version);
181 wprintf(L"Copyright (C) Microsoft Corp.\n");
182 #ifdef _DEBUG
183 wprintf(L"*** Debug build ***\n");
184 #endif

189 void SearchForFiles(const std::filesystem::path& path, std::list<SConversion>& files, bool recursive, _In_opt_z_ const wchar_t* folder)
190 {
191 // Process files
192 WIN32_FIND_DATAW findData = {};
193 ScopedFindHandle hFile(safe_handle(FindFirstFileExW(path.c_str(),
194 FindExInfoBasic, &findData,
195 FindExSearchNameMatch, nullptr,
196 FIND_FIRST_EX_LARGE_FETCH)));
197 if (hFile)
198 {
199 for (;;)
200 {
201 if (!(findData.dwFileAttributes & (FILE_ATTRIBUTE_HIDDEN | FILE_ATTRIBUTE_SYSTEM | FILE_ATTRIBUTE_DIRECTORY)))
202 {
203 SConversion conv = {};
204 conv.szSrc = path.parent_path().append(findData.cFileName).native();
205 if (folder)
206 {
207 conv.szFolder = folder;
208 }
209 files.push_back(conv);
210 }
211
212 if (!FindNextFileW(hFile.get(), &findData))
213 break;
214 }
215 }
216
217 // Process directories
218 if (recursive)
219 {
220 auto searchDir = path.parent_path().append(L"*");
221
222 hFile.reset(safe_handle(FindFirstFileExW(searchDir.c_str(),
223 FindExInfoBasic, &findData,
224 FindExSearchLimitToDirectories, nullptr,
225 FIND_FIRST_EX_LARGE_FETCH)));
226 if (!hFile)
227 return;
228
229 for (;;)
230 {
231 if (findData.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY)
232 {
233 if (findData.cFileName[0] != L'.')
234 {
235 auto subfolder = (folder)
236 ? (std::wstring(folder) + std::wstring(findData.cFileName) + std::filesystem::path::preferred_separator)
237 : (std::wstring(findData.cFileName) + std::filesystem::path::preferred_separator);
238
239 auto subdir = path.parent_path().append(findData.cFileName).append(path.filename().c_str());
240
241 SearchForFiles(subdir, files, recursive, subfolder.c_str());
242 }
243 }
244
245 if (!FindNextFileW(hFile.get(), &findData))
246 break;
247 }
248 }

251 void ProcessFileList(std::wifstream& inFile, std::list<SConversion>& files)
252 {
253 std::list<SConversion> flist;
254 std::set<std::wstring> excludes;
255
256 for (;;)
257 {
258 std::wstring fname;
259 std::getline(inFile, fname);
260 if (!inFile)
261 break;
262
263 if (fname[0] == L'#')
264 {
265 // Comment
266 }
267 else if (fname[0] == L'-')
268 {
269 if (flist.empty())
270 {
271 wprintf(L"WARNING: Ignoring the line '%ls' in -flist\n", fname.c_str());
272 }
273 else
274 {
275 std::filesystem::path path(fname.c_str() + 1);
276 auto& npath = path.make_preferred();
277 if (wcspbrk(fname.c_str(), L"?*") != nullptr)
278 {
279 std::list<SConversion> removeFiles;
280 SearchForFiles(npath, removeFiles, false, nullptr);
281
282 for (auto& it : removeFiles)
283 {
284 std::wstring name = it.szSrc;
285 std::transform(name.begin(), name.end(), name.begin(), towlower);
286 excludes.insert(name);
287 }
288 }
289 else
290 {
291 std::wstring name = npath.c_str();
292 std::transform(name.begin(), name.end(), name.begin(), towlower);
293 excludes.insert(name);
294 }
295 }
296 }
297 else if (wcspbrk(fname.c_str(), L"?*") != nullptr)
298 {
299 std::filesystem::path path(fname.c_str());
300 SearchForFiles(path.make_preferred(), flist, false, nullptr);
301 }
302 else
303 {
304 SConversion conv = {};
305 std::filesystem::path path(fname.c_str());
306 conv.szSrc = path.make_preferred().native();
307 flist.push_back(conv);
308 }
309 }
310
311 inFile.close();
312
313 if (!excludes.empty())
314 {
315 // Remove any excluded files
316 for (auto it = flist.begin(); it != flist.end();)
317 {
318 std::wstring name = it->szSrc;
319 std::transform(name.begin(), name.end(), name.begin(), towlower);
320 auto item = it;
321 ++it;
322 if (excludes.find(name) != excludes.end())
323 {
324 flist.erase(item);
325 }
326 }
327 }
328
329 if (flist.empty())
330 {
331 wprintf(L"WARNING: No file names found in -flist\n");
332 }
333 else
334 {
335 files.splice(files.end(), flist);
336 }
337 }

339 const wchar_t* GetErrorDesc(HRESULT hr) noexcept
340 {
341 static wchar_t desc[1024] = {};
342
343 LPWSTR errorText = nullptr;
344
345 const DWORD result = FormatMessageW(FORMAT_MESSAGE_FROM_SYSTEM | FORMAT_MESSAGE_IGNORE_INSERTS | FORMAT_MESSAGE_ALLOCATE_BUFFER,
346 nullptr, static_cast<DWORD>(hr),
347 MAKELANGID(LANG_NEUTRAL, SUBLANG_DEFAULT), reinterpret_cast<LPWSTR>(&errorText), 0, nullptr);
348
349 *desc = 0;
350
351 if (result > 0 && errorText)
352 {
353 swprintf_s(desc, L": %ls", errorText);
354
355 size_t len = wcslen(desc);
356 if (len >= 1)
357 {
358 desc[len - 1] = 0;
359 }
360
361 if (errorText)
362 LocalFree(errorText);
363
364 for (wchar_t* ptr = desc; *ptr != 0; ++ptr)
365 {
366 if (*ptr == L'\r' || *ptr == L'\n')
367 {
368 *ptr = L' ';
369 }
370 }
371 }
372
373 return desc;
374 }
