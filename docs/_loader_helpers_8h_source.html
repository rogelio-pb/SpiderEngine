324 inline HRESULT LoadTextureDataFromMemory(
325 _In_reads_(ddsDataSize) const uint8_t* ddsData,
326 size_t ddsDataSize,
327 const DDS_HEADER** header,
328 const uint8_t** bitData,
329 size_t* bitSize) noexcept
330 {
331 if (!header || !bitData || !bitSize)
332 {
333 return E_POINTER;
334 }
335
336 *bitSize = 0;
337
338 if (ddsDataSize > UINT32_MAX)
339 {
340 return E_FAIL;
341 }
342
343 if (ddsDataSize < DDS_MIN_HEADER_SIZE)
344 {
345 return E_FAIL;
346 }
347
348 // DDS files always start with the same magic number ("DDS ")
349 const auto dwMagicNumber = *reinterpret_cast<const uint32_t*>(ddsData);
350 if (dwMagicNumber != DDS_MAGIC)
351 {
352 return E_FAIL;
353 }
354
355 auto hdr = reinterpret_cast<const DDS_HEADER*>(ddsData + sizeof(uint32_t));
356
357 // Verify header to validate DDS file
358 if (hdr->size != sizeof(DDS_HEADER) ||
359 hdr->ddspf.size != sizeof(DDS_PIXELFORMAT))
360 {
361 return E_FAIL;
362 }
363
364 // Check for DX10 extension
365 bool bDXT10Header = false;
366 if ((hdr->ddspf.flags & DDS_FOURCC) &&
367 (MAKEFOURCC('D', 'X', '1', '0') == hdr->ddspf.fourCC))
368 {
369 // Must be long enough for both headers and magic value
370 if (ddsDataSize < DDS_DX10_HEADER_SIZE)
371 {
372 return E_FAIL;
373 }
374
375 bDXT10Header = true;
376 }
377
378 // setup the pointers in the process request
379 *header = hdr;
380 auto offset = DDS_MIN_HEADER_SIZE
381 + (bDXT10Header ? sizeof(DDS_HEADER_DXT10) : 0u);
382 *bitData = ddsData + offset;
383 *bitSize = ddsDataSize - offset;
384
385 return S_OK;
386 }

388 //--------------------------------------------------------------------------------------
389 inline HRESULT LoadTextureDataFromFile(
390 _In_z_ const wchar_t* fileName,
391 std::unique_ptr<uint8_t[]>& ddsData,
392 const DDS_HEADER** header,
393 const uint8_t** bitData,
394 size_t* bitSize) noexcept
395 {
396 if (!header || !bitData || !bitSize)
397 {
398 return E_POINTER;
399 }
400
401 *bitSize = 0;
402
403 // open the file
404 ScopedHandle hFile(safe_handle(CreateFile2(
405 fileName,
406 GENERIC_READ, FILE_SHARE_READ, OPEN_EXISTING,
407 nullptr)));
408 if (!hFile)
409 {
410 return HRESULT_FROM_WIN32(GetLastError());
411 }
412
413 // Get the file size
414 FILE_STANDARD_INFO fileInfo;
415 if (!GetFileInformationByHandleEx(hFile.get(), FileStandardInfo, &fileInfo, sizeof(fileInfo)))
416 {
417 return HRESULT_FROM_WIN32(GetLastError());
418 }
419
420 // File is too big for 32-bit allocation, so reject read
421 if (fileInfo.EndOfFile.HighPart > 0)
422 {
423 return E_FAIL;
424 }
425
426 // Need at least enough data to fill the header and magic number to be a valid DDS
427 if (fileInfo.EndOfFile.LowPart < DDS_MIN_HEADER_SIZE)
428 {
429 return E_FAIL;
430 }
431
432 // create enough space for the file data
433 ddsData.reset(new (std::nothrow) uint8_t[fileInfo.EndOfFile.LowPart]);
434 if (!ddsData)
435 {
436 return E_OUTOFMEMORY;
437 }
438
439 // read the data in
440 DWORD bytesRead = 0;
441 if (!ReadFile(hFile.get(),
442 ddsData.get(),
443 fileInfo.EndOfFile.LowPart,
444 &bytesRead,
445 nullptr
446 ))
447 {
448 ddsData.reset();
449 return HRESULT_FROM_WIN32(GetLastError());
450 }
451
452 if (bytesRead < fileInfo.EndOfFile.LowPart)
453 {
454 ddsData.reset();
455 return E_FAIL;
456 }
457
458 // DDS files always start with the same magic number ("DDS ")
459 const auto dwMagicNumber = *reinterpret_cast<const uint32_t*>(ddsData.get());
460 if (dwMagicNumber != DDS_MAGIC)
461 {
462 ddsData.reset();
463 return E_FAIL;
464 }
465
466 auto hdr = reinterpret_cast<const DDS_HEADER*>(ddsData.get() + sizeof(uint32_t));
467
468 // Verify header to validate DDS file
469 if (hdr->size != sizeof(DDS_HEADER) ||
470 hdr->ddspf.size != sizeof(DDS_PIXELFORMAT))
471 {
472 ddsData.reset();
473 return E_FAIL;
474 }
475
476 // Check for DX10 extension
477 bool bDXT10Header = false;
478 if ((hdr->ddspf.flags & DDS_FOURCC) &&
479 (MAKEFOURCC('D', 'X', '1', '0') == hdr->ddspf.fourCC))
480 {
481 // Must be long enough for both headers and magic value
482 if (fileInfo.EndOfFile.LowPart < DDS_DX10_HEADER_SIZE)
483 {
484 ddsData.reset();
485 return E_FAIL;
486 }
487
488 bDXT10Header = true;
489 }
490
491 // setup the pointers in the process request
492 *header = hdr;
493 auto offset = DDS_MIN_HEADER_SIZE
494 + (bDXT10Header ? sizeof(DDS_HEADER_DXT10) : 0u);
495 *bitData = ddsData.get() + offset;
496 *bitSize = fileInfo.EndOfFile.LowPart - offset;
497
498 return S_OK;
499 }

501 //--------------------------------------------------------------------------------------
502 // Get surface information for a particular format
503 //--------------------------------------------------------------------------------------
504 inline HRESULT GetSurfaceInfo(
505 _In_ size_t width,
506 _In_ size_t height,
507 _In_ DXGI_FORMAT fmt,
508 _Out_opt_ size_t* outNumBytes,
509 _Out_opt_ size_t* outRowBytes,
510 _Out_opt_ size_t* outNumRows) noexcept
511 {
512 uint64_t numBytes = 0;
513 uint64_t rowBytes = 0;
514 uint64_t numRows = 0;
515
516 bool bc = false;
517 bool packed = false;
518 bool planar = false;
519 size_t bpe = 0;
520 switch (fmt)
521 {
522 case DXGI_FORMAT_UNKNOWN:
523 return E_INVALIDARG;
524
525 case DXGI_FORMAT_BC1_TYPELESS:
526 case DXGI_FORMAT_BC1_UNORM:
527 case DXGI_FORMAT_BC1_UNORM_SRGB:
528 case DXGI_FORMAT_BC4_TYPELESS:
529 case DXGI_FORMAT_BC4_UNORM:
530 case DXGI_FORMAT_BC4_SNORM:
531 bc = true;
532 bpe = 8;
533 break;
534
535 case DXGI_FORMAT_BC2_TYPELESS:
536 case DXGI_FORMAT_BC2_UNORM:
537 case DXGI_FORMAT_BC2_UNORM_SRGB:
538 case DXGI_FORMAT_BC3_TYPELESS:
539 case DXGI_FORMAT_BC3_UNORM:
540 case DXGI_FORMAT_BC3_UNORM_SRGB:
541 case DXGI_FORMAT_BC5_TYPELESS:
542 case DXGI_FORMAT_BC5_UNORM:
543 case DXGI_FORMAT_BC5_SNORM:
544 case DXGI_FORMAT_BC6H_TYPELESS:
545 case DXGI_FORMAT_BC6H_UF16:
546 case DXGI_FORMAT_BC6H_SF16:
547 case DXGI_FORMAT_BC7_TYPELESS:
548 case DXGI_FORMAT_BC7_UNORM:
549 case DXGI_FORMAT_BC7_UNORM_SRGB:
550 bc = true;
551 bpe = 16;
552 break;
553
554 case DXGI_FORMAT_R8G8_B8G8_UNORM:
555 case DXGI_FORMAT_G8R8_G8B8_UNORM:
556 case DXGI_FORMAT_YUY2:
557 packed = true;
558 bpe = 4;
559 break;
560
561 case DXGI_FORMAT_Y210:
562 case DXGI_FORMAT_Y216:
563 packed = true;
564 bpe = 8;
565 break;
566
567 case DXGI_FORMAT_NV12:
568 case DXGI_FORMAT_420_OPAQUE:
569 if ((height % 2) != 0)
570 {
571 // Requires a height alignment of 2.
572 return E_INVALIDARG;
573 }
574 planar = true;
575 bpe = 2;
576 break;
577
578 #if (_WIN32_WINNT >= _WIN32_WINNT_WIN10)
579
580 case DXGI_FORMAT_P208:
581 planar = true;
582 bpe = 2;
583 break;
584
585 #endif
586
587 case DXGI_FORMAT_P010:
588 case DXGI_FORMAT_P016:
589 if ((height % 2) != 0)
590 {
591 // Requires a height alignment of 2.
592 return E_INVALIDARG;
593 }
594 planar = true;
595 bpe = 4;
596 break;
597
598 #if (defined(_XBOX_ONE) && defined(_TITLE)) || defined(_GAMING_XBOX)
599
600 case DXGI_FORMAT_D16_UNORM_S8_UINT:
601 case DXGI_FORMAT_R16_UNORM_X8_TYPELESS:
602 case DXGI_FORMAT_X16_TYPELESS_G8_UINT:
603 planar = true;
604 bpe = 4;
605 break;
606
607 #endif
608
609 default:
610 break;
611 }
612
613 if (bc)
614 {
615 uint64_t numBlocksWide = 0;
616 if (width > 0)
617 {
618 numBlocksWide = std::max<uint64_t>(1u, (uint64_t(width) + 3u) / 4u);
619 }
620 uint64_t numBlocksHigh = 0;
621 if (height > 0)
622 {
623 numBlocksHigh = std::max<uint64_t>(1u, (uint64_t(height) + 3u) / 4u);
624 }
625 rowBytes = numBlocksWide * bpe;
626 numRows = numBlocksHigh;
627 numBytes = rowBytes * numBlocksHigh;
628 }
629 else if (packed)
630 {
631 rowBytes = ((uint64_t(width) + 1u) >> 1) * bpe;
632 numRows = uint64_t(height);
633 numBytes = rowBytes * height;
634 }
635 else if (fmt == DXGI_FORMAT_NV11)
636 {
637 rowBytes = ((uint64_t(width) + 3u) >> 2) * 4u;
638 numRows = uint64_t(height) * 2u; // Direct3D makes this simplifying assumption, although it is larger than the 4:1:1 data
639 numBytes = rowBytes * numRows;
640 }
641 else if (planar)
642 {
643 rowBytes = ((uint64_t(width) + 1u) >> 1) * bpe;
644 numBytes = (rowBytes * uint64_t(height)) + ((rowBytes * uint64_t(height) + 1u) >> 1);
645 numRows = height + ((uint64_t(height) + 1u) >> 1);
646 }
647 else
648 {
649 const size_t bpp = BitsPerPixel(fmt);
650 if (!bpp)
651 return E_INVALIDARG;
652
653 rowBytes = (uint64_t(width) * bpp + 7u) / 8u; // round up to nearest byte
654 numRows = uint64_t(height);
655 numBytes = rowBytes * height;
656 }
657
658 #if defined(_M_IX86) || defined(_M_ARM) || defined(_M_HYBRID_X86_ARM64)
659 static_assert(sizeof(size_t) == 4, "Not a 32-bit platform!");
660 if (numBytes > UINT32_MAX || rowBytes > UINT32_MAX || numRows > UINT32_MAX)
661 return HRESULT_FROM_WIN32(ERROR_ARITHMETIC_OVERFLOW);
662 #else
663 static_assert(sizeof(size_t) == 8, "Not a 64-bit platform!");
664 #endif
665
666 if (outNumBytes)
667 {
668 *outNumBytes = static_cast<size_t>(numBytes);
669 }
670 if (outRowBytes)
671 {
672 *outRowBytes = static_cast<size_t>(rowBytes);
673 }
674 if (outNumRows)
675 {
676 *outNumRows = static_cast<size_t>(numRows);
677 }
678
679 return S_OK;
680 }

683 #define ISBITMASK( r,g,b,a ) ( ddpf.RBitMask == r && ddpf.GBitMask == g && ddpf.BBitMask == b && ddpf.ABitMask == a )
684
685 inline DXGI_FORMAT GetDXGIFormat(const DDS_PIXELFORMAT& ddpf) noexcept
686 {
687 if (ddpf.flags & DDS_RGB)
688 {
689 // Note that sRGB formats are written using the "DX10" extended header
690
691 switch (ddpf.RGBBitCount)
692 {
693 case 32:
694 if (ISBITMASK(0x000000ff, 0x0000ff00, 0x00ff0000, 0xff000000))
695 {
696 return DXGI_FORMAT_R8G8B8A8_UNORM;
697 }
698
699 if (ISBITMASK(0x00ff0000, 0x0000ff00, 0x000000ff, 0xff000000))
700 {
701 return DXGI_FORMAT_B8G8R8A8_UNORM;
702 }
703
704 if (ISBITMASK(0x00ff0000, 0x0000ff00, 0x000000ff, 0))
705 {
706 return DXGI_FORMAT_B8G8R8X8_UNORM;
707 }
708
709 // No DXGI format maps to ISBITMASK(0x000000ff,0x0000ff00,0x00ff0000,0) aka D3DFMT_X8B8G8R8
710
711 // Note that many common DDS reader/writers (including D3DX) swap the
712 // the RED/BLUE masks for 10:10:10:2 formats. We assume
713 // below that the 'backwards' header mask is being used since it is most
714 // likely written by D3DX. The more robust solution is to use the 'DX10'
715 // header extension and specify the DXGI_FORMAT_R10G10B10A2_UNORM format directly
716
717 // For 'correct' writers, this should be 0x000003ff,0x000ffc00,0x3ff00000 for RGB data
718 if (ISBITMASK(0x3ff00000, 0x000ffc00, 0x000003ff, 0xc0000000))
719 {
720 return DXGI_FORMAT_R10G10B10A2_UNORM;
721 }
722
723 // No DXGI format maps to ISBITMASK(0x000003ff,0x000ffc00,0x3ff00000,0xc0000000) aka D3DFMT_A2R10G10B10
724
725 if (ISBITMASK(0x0000ffff, 0xffff0000, 0, 0))
726 {
727 return DXGI_FORMAT_R16G16_UNORM;
728 }
729
730 if (ISBITMASK(0xffffffff, 0, 0, 0))
731 {
732 // Only 32-bit color channel format in D3D9 was R32F
733 return DXGI_FORMAT_R32_FLOAT; // D3DX writes this out as a FourCC of 114
734 }
735 break;
736
737 case 24:
738 // No 24bpp DXGI formats aka D3DFMT_R8G8B8
739 break;
740
741 case 16:
742 if (ISBITMASK(0x7c00, 0x03e0, 0x001f, 0x8000))
743 {
744 return DXGI_FORMAT_B5G5R5A1_UNORM;
745 }
746 if (ISBITMASK(0xf800, 0x07e0, 0x001f, 0))
747 {
748 return DXGI_FORMAT_B5G6R5_UNORM;
749 }
750
751 // No DXGI format maps to ISBITMASK(0x7c00,0x03e0,0x001f,0) aka D3DFMT_X1R5G5B5
752
753 if (ISBITMASK(0x0f00, 0x00f0, 0x000f, 0xf000))
754 {
755 return DXGI_FORMAT_B4G4R4A4_UNORM;
756 }
757
758 // NVTT versions 1.x wrote this as RGB instead of LUMINANCE
759 if (ISBITMASK(0x00ff, 0, 0, 0xff00))
760 {
761 return DXGI_FORMAT_R8G8_UNORM;
762 }
763 if (ISBITMASK(0xffff, 0, 0, 0))
764 {
765 return DXGI_FORMAT_R16_UNORM;
766 }
767
768 // No DXGI format maps to ISBITMASK(0x0f00,0x00f0,0x000f,0) aka D3DFMT_X4R4G4B4
769
770 // No 3:3:2:8 or paletted DXGI formats aka D3DFMT_A8R3G3B2, D3DFMT_A8P8, etc.
771 break;
772
773 case 8:
774 // NVTT versions 1.x wrote this as RGB instead of LUMINANCE
775 if (ISBITMASK(0xff, 0, 0, 0))
776 {
777 return DXGI_FORMAT_R8_UNORM;
778 }
779
780 // No 3:3:2 or paletted DXGI formats aka D3DFMT_R3G3B2, D3DFMT_P8
781 break;
782
783 default:
784 return DXGI_FORMAT_UNKNOWN;
785 }
786 }
787 else if (ddpf.flags & DDS_LUMINANCE)
788 {
789 switch (ddpf.RGBBitCount)
790 {
791 case 16:
792 if (ISBITMASK(0xffff, 0, 0, 0))
793 {
794 return DXGI_FORMAT_R16_UNORM; // D3DX10/11 writes this out as DX10 extension
795 }
796 if (ISBITMASK(0x00ff, 0, 0, 0xff00))
797 {
798 return DXGI_FORMAT_R8G8_UNORM; // D3DX10/11 writes this out as DX10 extension
799 }
800 break;
801
802 case 8:
803 if (ISBITMASK(0xff, 0, 0, 0))
804 {
805 return DXGI_FORMAT_R8_UNORM; // D3DX10/11 writes this out as DX10 extension
806 }
807
808 // No DXGI format maps to ISBITMASK(0x0f,0,0,0xf0) aka D3DFMT_A4L4
809
810 if (ISBITMASK(0x00ff, 0, 0, 0xff00))
811 {
812 return DXGI_FORMAT_R8G8_UNORM; // Some DDS writers assume the bitcount should be 8 instead of 16
813 }
814 break;
815
816 default:
817 return DXGI_FORMAT_UNKNOWN;
818 }
819 }
820 else if (ddpf.flags & DDS_ALPHA)
821 {
822 if (8 == ddpf.RGBBitCount)
823 {
824 return DXGI_FORMAT_A8_UNORM;
825 }
826 }
827 else if (ddpf.flags & DDS_BUMPDUDV)
828 {
829 switch (ddpf.RGBBitCount)
830 {
831 case 32:
832 if (ISBITMASK(0x000000ff, 0x0000ff00, 0x00ff0000, 0xff000000))
833 {
834 return DXGI_FORMAT_R8G8B8A8_SNORM; // D3DX10/11 writes this out as DX10 extension
835 }
836 if (ISBITMASK(0x0000ffff, 0xffff0000, 0, 0))
837 {
838 return DXGI_FORMAT_R16G16_SNORM; // D3DX10/11 writes this out as DX10 extension
839 }
840
841 // No DXGI format maps to ISBITMASK(0x3ff00000, 0x000ffc00, 0x000003ff, 0xc0000000) aka D3DFMT_A2W10V10U10
842 break;
843
844 case 16:
845 if (ISBITMASK(0x00ff, 0xff00, 0, 0))
846 {
847 return DXGI_FORMAT_R8G8_SNORM; // D3DX10/11 writes this out as DX10 extension
848 }
849 break;
850
851 default:
852 return DXGI_FORMAT_UNKNOWN;
853 }
854
855 // No DXGI format maps to DDPF_BUMPLUMINANCE aka D3DFMT_L6V5U5, D3DFMT_X8L8V8U8
856 }
857 else if (ddpf.flags & DDS_FOURCC)
858 {
859 if (MAKEFOURCC('D', 'X', 'T', '1') == ddpf.fourCC)
860 {
861 return DXGI_FORMAT_BC1_UNORM;
862 }
863 if (MAKEFOURCC('D', 'X', 'T', '3') == ddpf.fourCC)
864 {
865 return DXGI_FORMAT_BC2_UNORM;
866 }
867 if (MAKEFOURCC('D', 'X', 'T', '5') == ddpf.fourCC)
868 {
869 return DXGI_FORMAT_BC3_UNORM;
870 }
871
872 // While pre-multiplied alpha isn't directly supported by the DXGI formats,
873 // they are basically the same as these BC formats so they can be mapped
874 if (MAKEFOURCC('D', 'X', 'T', '2') == ddpf.fourCC)
875 {
876 return DXGI_FORMAT_BC2_UNORM;
877 }
878 if (MAKEFOURCC('D', 'X', 'T', '4') == ddpf.fourCC)
879 {
880 return DXGI_FORMAT_BC3_UNORM;
881 }
882
883 if (MAKEFOURCC('A', 'T', 'I', '1') == ddpf.fourCC)
884 {
885 return DXGI_FORMAT_BC4_UNORM;
886 }
887 if (MAKEFOURCC('B', 'C', '4', 'U') == ddpf.fourCC)
888 {
889 return DXGI_FORMAT_BC4_UNORM;
890 }
891 if (MAKEFOURCC('B', 'C', '4', 'S') == ddpf.fourCC)
892 {
893 return DXGI_FORMAT_BC4_SNORM;
894 }
895
896 if (MAKEFOURCC('A', 'T', 'I', '2') == ddpf.fourCC)
897 {
898 return DXGI_FORMAT_BC5_UNORM;
899 }
900 if (MAKEFOURCC('B', 'C', '5', 'U') == ddpf.fourCC)
901 {
902 return DXGI_FORMAT_BC5_UNORM;
903 }
904 if (MAKEFOURCC('B', 'C', '5', 'S') == ddpf.fourCC)
905 {
906 return DXGI_FORMAT_BC5_SNORM;
907 }
908
909 // BC6H and BC7 are written using the "DX10" extended header
910
911 if (MAKEFOURCC('R', 'G', 'B', 'G') == ddpf.fourCC)
912 {
913 return DXGI_FORMAT_R8G8_B8G8_UNORM;
914 }
915 if (MAKEFOURCC('G', 'R', 'G', 'B') == ddpf.fourCC)
916 {
917 return DXGI_FORMAT_G8R8_G8B8_UNORM;
918 }
919
920 if (MAKEFOURCC('Y', 'U', 'Y', '2') == ddpf.fourCC)
921 {
922 return DXGI_FORMAT_YUY2;
923 }
924
925 // Check for D3DFORMAT enums being set here
926 switch (ddpf.fourCC)
927 {
928 case 36: // D3DFMT_A16B16G16R16
929 return DXGI_FORMAT_R16G16B16A16_UNORM;
930
931 case 110: // D3DFMT_Q16W16V16U16
932 return DXGI_FORMAT_R16G16B16A16_SNORM;
933
934 case 111: // D3DFMT_R16F
935 return DXGI_FORMAT_R16_FLOAT;
936
937 case 112: // D3DFMT_G16R16F
938 return DXGI_FORMAT_R16G16_FLOAT;
939
940 case 113: // D3DFMT_A16B16G16R16F
941 return DXGI_FORMAT_R16G16B16A16_FLOAT;
942
943 case 114: // D3DFMT_R32F
944 return DXGI_FORMAT_R32_FLOAT;
945
946 case 115: // D3DFMT_G32R32F
947 return DXGI_FORMAT_R32G32_FLOAT;
948
949 case 116: // D3DFMT_A32B32G32R32F
950 return DXGI_FORMAT_R32G32B32A32_FLOAT;
951
952 // No DXGI format maps to D3DFMT_CxV8U8
953
954 default:
955 return DXGI_FORMAT_UNKNOWN;
956 }
957 }
958
959 return DXGI_FORMAT_UNKNOWN;
960 }

1095 {
