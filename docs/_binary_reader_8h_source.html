44 // Reads an array of values.
45 template<typename T> T const* ReadArray(size_t elementCount)
46 {
47 static_assert(std::is_standard_layout<T>::value, "Can only read plain-old-data types");
48
49 const uint64_t byteCount = uint64_t(sizeof(T)) * uint64_t(elementCount);
50 if (byteCount > UINT32_MAX)
51 throw std::overflow_error("ReadArray");
52
53 uint8_t const* newPos = mPos + static_cast<size_t>(byteCount);
54
55 if (newPos < mPos)
56 throw std::overflow_error("ReadArray");
57
58 if (newPos > mEnd)
59 throw std::runtime_error("End of file");
60
61 auto result = reinterpret_cast<T const*>(mPos);
62
63 mPos = newPos;
64
65 return result;
66 }
